## 1. 설계 철학

### 1.1 컴포넌트 분류 (Atomic Design 변형)

| 계층 | 위치 | 책임 | 예시 |
|---|---|---|---|
| **UI Primitives** | `components/ui/` | 원자 컴포넌트 (재사용) | `Button`, `Input`, `Modal` |
| **Composite** | `components/` | 여러 primitive 조합 | `NoteCard`, `FolderTree` |
| **Feature** | `features/<domain>/` | 도메인 로직 + 상태 결합 | `TiptapEditor`, `GraphCanvas` |
| **Page** | `app/**/page.tsx` | 데이터 페칭 + 4-pane 조립 | `NotesPage` |

> 💡 **규칙**: 상위 계층은 하위만 import. UI Primitives는 외부 의존 0.

### 1.2 상태 관리 전략

| 종류 | 도구 | 우리 사례 |
|---|---|---|
| **서버 상태** (캐시·revalidate) | **SWR** | 노트 목록, 그래프 데이터, 자주 쓴 태그, 캘린더 |
| **전역 클라이언트 상태** | **Zustand** | 인증 사용자, **멀티탭**, 현재 폴더, 테마, 모드(파일/그래프) |
| **폼 상태** | **react-hook-form + zod** | 로그인·회원가입·노트 메타 |
| **로컬 UI 상태** | `useState` | 드롭다운 열림, 호버, 토글 |

> 🚫 **금지**: 서버 상태를 Zustand에 저장 X (캐시 무효화 지옥). SWR이 캐시 담당.

### 5.1 TiptapEditor — Tiptap 에디터 + #@& 배지 + 표 (FE1, v2 핵심)

> Tiptap은 ProseMirror 기반 WYSIWYG. 마크다운 입출력 + 인라인 노드(배지) + 표 GUI를 한 패키지로 지원.

```typescript
// features/editor/TiptapEditor.tsx
import { useEditor, EditorContent } from '@tiptap/react';
import StarterKit from '@tiptap/starter-kit';
import Table from '@tiptap/extension-table';
import TableRow from '@tiptap/extension-table-row';
import TableCell from '@tiptap/extension-table-cell';
import TableHeader from '@tiptap/extension-table-header';
import Mention from '@tiptap/extension-mention';
import { Markdown } from 'tiptap-markdown';
import { TokenBadge } from './extensions/TokenBadge';   // 직접 만든 inline node
import { useAutosave } from '@/hooks/useAutosave';
import { useAutocomplete } from '@/hooks/useAutocomplete';

type TiptapEditorProps = {
  noteId: string;
  initialContent: string;
};

export function TiptapEditor({ noteId, initialContent }: TiptapEditorProps) {
  const editor = useEditor({
    extensions: [
      StarterKit,
      Markdown,
      Table.configure({ resizable: true }),
      TableRow, TableHeader, TableCell,
      TokenBadge.configure({ types: ['mention', 'tag', 'object'] }),
      Mention.configure({
        suggestion: useAutocomplete(),   // #@& 자동완성 훅
      }),
    ],
    content: initialContent,
    onUpdate: ({ editor }) => {
      const markdown = editor.storage.markdown.getMarkdown();
      onContentChange(markdown);
    },
  });

  // 자동저장 (실시간)
  useAutosave(noteId, editor);

  return (
    <div className="flex h-full flex-col">
      <Toolbar editor={editor}/>
      <EditorContent editor={editor} className="prose max-w-none p-4"/>
    </div>
  );
}
```

#### 5.1.1 TokenBadge 확장 (`#@&` 인라인 노드)

```typescript
// features/editor/extensions/TokenBadge.ts
import { Node, mergeAttributes } from '@tiptap/core';

export const TokenBadge = Node.create({
  name: 'tokenBadge',
  group: 'inline',
  inline: true,
  atom: true,
  addAttributes() {
    return {
      type: { default: 'tag' },   // 'mention' | 'tag' | 'object'
      label: { default: '' },
    };
  },
  parseHTML() {
    return [{ tag: 'span[data-token-badge]' }];
  },
  renderHTML({ HTMLAttributes }) {
    const colors = { mention: 'bg-blue-100 text-blue-700', tag: 'bg-emerald-100 text-emerald-700', object: 'bg-red-100 text-red-700' };
    const prefix = { mention: '@', tag: '#', object: '&' };
    return [
      'span',
      mergeAttributes(HTMLAttributes, {
        'data-token-badge': '',
        class: `rounded px-1 ${colors[HTMLAttributes.type]}`,
      }),
      `${prefix[HTMLAttributes.type]}${HTMLAttributes.label}`,
    ];
  },
});
```

#### 5.1.2 useAutosave 훅 (모든 탭 실시간 저장)

```typescript
// hooks/useAutosave.ts
import { useEffect, useRef } from 'react';
import { useDebounce } from './useDebounce';
import { notesApi } from '@/lib/api/notes';
import { useTabStore } from '@/store/tabStore';

export function useAutosave(noteId: string, editor: Editor | null) {
  const setSaveState = useTabStore((s) => s.setSaveState);
  const content = editor?.storage.markdown.getMarkdown() ?? '';
  const debouncedContent = useDebounce(content, 2000);

  useEffect(() => {
    if (!editor || debouncedContent === '') return;
    setSaveState(noteId, 'saving');
    notesApi.update(noteId, { content: debouncedContent })
      .then(() => setSaveState(noteId, 'saved'))
      .catch(() => setSaveState(noteId, 'error'));
  }, [debouncedContent, noteId, editor]);

  // 페이지 종료 / 탭 닫기 시 즉시 flush
  useEffect(() => {
    const handler = (e: BeforeUnloadEvent) => {
      if (content !== debouncedContent) {
        notesApi.update(noteId, { content });   // 동기 호출 (best effort)
      }
    };
    window.addEventListener('beforeunload', handler);
    return () => window.removeEventListener('beforeunload', handler);
  }, [noteId, content, debouncedContent]);
}
```

```typescript
import ForceGraph2D from 'react-force-graph-2d';
import { useCallback, useRef } from 'react';

type GraphCanvasProps = {
  data: {
    nodes: GraphNode[];
    links: GraphLink[];
  };
  onNodeHover: (node: GraphNode | null) => void;
  onNodeDoubleClick: (node: GraphNode) => void;
};

export function GraphCanvas({ data, onNodeHover, onNodeDoubleClick }: GraphCanvasProps) {
  const fgRef = useRef<any>();

  return (
    <ForceGraph2D
      ref={fgRef}
      graphData={data}
      nodeLabel="label"
      nodeColor={(n: any) => n.color}
      nodeVal={(n: any) => n.size}
      linkColor={() => '#888'}
      linkWidth={(l: any) => Math.sqrt(l.weight)}
      linkDirectionalArrowLength={(l: any) => l.type === 'contains' ? 3 : 0}
      onNodeHover={onNodeHover}
      onNodeClick={(node: any) => onNodeDoubleClick(node)}   // 2D는 더블클릭 대신 클릭
    />
  );
}
```

```typescript
  activeTabId: string | null;
  open: (noteId: string) => void;       // 이미 있으면 활성화, 없으면 추가
  activate: (tabId: string) => void;
  close: (tabId: string) => void;       // 자동저장 flush 후 닫기
  setSaveState: (tabId: string, state: Tab['saveState']) => void;
  hydrate: () => void;                  // localStorage 복원
};

export const useTabStore = create<TabState>((set, get) => ({
  tabs: [],
  activeTabId: null,
  open: (noteId) => {
    const existing = get().tabs.find((t) => t.id === noteId);
    if (existing) {
      set({ activeTabId: noteId });
    } else {
      const newTab: Tab = { id: noteId, title: '...', saveState: 'idle' };
      set((s) => ({ tabs: [...s.tabs, newTab], activeTabId: noteId }));
      persistTabs(get().tabs);
    }
  },
  activate: (tabId) => set({ activeTabId: tabId }),
  close: (tabId) => {
    set((s) => {
      const newTabs = s.tabs.filter((t) => t.id !== tabId);
      const newActive = s.activeTabId === tabId
        ? (newTabs[newTabs.length - 1]?.id ?? null)
        : s.activeTabId;
      return { tabs: newTabs, activeTabId: newActive };
    });
    persistTabs(get().tabs);
  },
  setSaveState: (tabId, state) => set((s) => ({
    tabs: s.tabs.map((t) => t.id === tabId ? { ...t, saveState: state } : t),
  })),
  hydrate: () => {
    const saved = JSON.parse(localStorage.getItem('tabs') ?? '[]');
    set({ tabs: saved, activeTabId: saved[0]?.id ?? null });
  },
}));

function persistTabs(tabs: Tab[]) {
  localStorage.setItem('tabs', JSON.stringify(tabs.map(({ id, title }) => ({ id, title }))));
}
```
